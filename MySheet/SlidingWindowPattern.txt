Fixed size Sliding window 
Template:

int left = 0;
int sum = 0;
int ans = 0;

for(int right = 0; right < nums.size(); right++) {

    sum += nums[right];

    if(right - left + 1 == k) {

        ans = max(ans, sum);

        sum -= nums[left];
        left++;
    }
}
